## Functional smoke: record_message + parse_message

```bash
set -a; . ./.env; set +a
.venv/bin/python -c "
import sys, json
sys.path.insert(0, 'src')
from services.base import BaseService
from services.llm.aimessage import AIMessage
s = BaseService()
session = s.db.create_session('test-base')
msg_id = s.record_message('human_input', 'hello', session, raw_response=json.dumps({'greeting': 'hi'}), num_tries=1)
print('recorded', msg_id)
aim = AIMessage(session_uuid=session, message_type_slug='human_input', unique_prompt='hello', raw_response=json.dumps({'greeting': 'hi'}))
parsed = s.parse_message(aim, 'human_input')
print('parsed', parsed)
"
```
